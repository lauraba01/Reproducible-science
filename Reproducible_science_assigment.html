df_newbird <-
  tibble(Airspeed=seq(min(df_bird1$Airspeed), #minimum of the range
         max(df_bird1$Airspeed), #max of the range,
         length=100)) #this tells R how many values to generate
df_pred <- predict(lm_wba_1,
                   newdata=df_newbird,
                   int ="c")
df_newbirds_pred <- bind_cols(df_newbird, as_tibble(df_pred))
